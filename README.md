def number_to_words(n):
    """
    Переводит число от 0 до 99 в слова на русском языке.
    
    Параметры:
    n (int): Целое число от 0 до 99.
    
    Возвращает:
    str: Число прописью.
    """
    # Словарь для чисел 0-19
    units = {
        0: 'ноль', 1: 'один', 2: 'два', 3: 'три', 4: 'четыре', 5: 'пять',
        6: 'шесть', 7: 'семь', 8: 'восемь', 9: 'девять', 10: 'десять',
        11: 'одиннадцать', 12: 'двенадцать', 13: 'тринадцать', 14: 'четырнадцать',
        15: 'пятнадцать', 16: 'шестнадцать', 17: 'семнадцать', 18: 'восемнадцать', 19: 'девятнадцать'
    }
    
    # Словарь для десятков (20, 30, ..., 90)
    tens = {
        2: 'двадцать', 3: 'тридцать', 4: 'сорок', 5: 'пятьдесят',
        6: 'шестьдесят', 7: 'семьдесят', 8: 'восемьдесят', 9: 'девяносто'
    }
    
    if n < 20:
        return units[n]  # Для чисел 0-19 берём готовое значение
    else:
        tens_digit = n // 10  # Целая часть (десятки)
        units_digit = n % 10   # Остаток (единицы)
        if units_digit == 0:
            return tens[tens_digit]  # Пример: 30 -> "тридцать"
        else:
            # Пример: 45 -> "сорок пять"
            return tens[tens_digit] + " " + units[units_digit]

# Пример использования
print(number_to_words(0))     # "ноль"
print(number_to_words(11))    # "одиннадцать"
print(number_to_words(30))    # "тридцать"
print(number_to_words(77))    # "семьдесят семь"
